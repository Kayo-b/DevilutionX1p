cmake_minimum_required(VERSION 3.16)
project(d1cli LANGUAGES CXX)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
set(D1GT_DIR "$ENV{HOME}/projects/d1-graphics-tool" CACHE PATH "d1-graphics-tool checkout")
set(T ${D1GT_DIR})
set(CMAKE_AUTOUIC_SEARCH_PATHS ${T}/source/dialogs ${T}/source/views ${T}/source/widgets ${T}/source)
file(GLOB SRCS ${T}/source/*/*.cpp ${T}/source/mainwindow.cpp ${T}/source/mainwindow.ui ${T}/source/*/*.ui)
add_executable(d1cli main.cpp ${SRCS} ${T}/resources/d1files.qrc)
target_include_directories(d1cli PRIVATE ${T}/source)
target_link_libraries(d1cli PRIVATE Qt6::Widgets)
